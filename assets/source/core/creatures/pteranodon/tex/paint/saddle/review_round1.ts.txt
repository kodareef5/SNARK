// The painted pteranodon drawn by the game's own renderer from a pack look.sh built: whole from below against the sky
// at the distances it is seen, from above and three quarters, and close where a rider looks (the head, the neck, the
// inner wing), in the design's dyes and in loud and dark ones. The renderer reads the rgba variant, so maps are at most
// 1,024 across here. It stands in its rest pose, the glide.
//
//   bun tex/paint/look.ts --source DIR --out DIR [--stages]
//
// With --stages, and hurt and decay layers named in the asset, it draws instead each view at every stage, unhurt, hurt
// 1 to 3 (3 is also the fresh corpse) and decay 1 to 3, a row each (stages_<view>_<dyes>.png).

// Made again from itself and the asset as it is now, so the layers the asset names are in it without a build.

/** The body and its worn saddle as one mesh of their finest levels, so one draw shows the animal saddled. */
function saddledPack(): Pack | null {
  const meshes = unpackMeshes(pack);
  const body = meshes.find((m) => m.id === "core:pteranodon")!;
  const saddle = meshes.find((m) => m.id.endsWith("/worn"));
  if (!saddle || !body.materials || !saddle.materials) return null;
  const [a, b] = [body.lods[0]!, saddle.lods[0]!];
  const count = a.positions.length / 3;
  const join = <T extends Float32Array | Uint8Array | Uint32Array>(x: T | undefined, y: T | undefined, make: (n: number) => T): T | undefined => {
    if (!x || !y) return undefined;
    const out = make(x.length + y.length);
    out.set(x);
    out.set(y, x.length);
    return out;
  };
  const lod: SourceLod = {
    positions: join(a.positions, b.positions, (n) => new Float32Array(n))!,
    colors: join(a.colors, b.colors, (n) => new Uint8Array(n))!,
    indices: Uint32Array.from([...a.indices, ...[...b.indices].map((i) => i + count)]),
    bones: join(a.bones, b.bones, (n) => new Uint8Array(n)),
    uvs: join(a.uvs, b.uvs, (n) => new Float32Array(n)),
    normals: join(a.normals, b.normals, (n) => new Float32Array(n)),
    tangents: join(a.tangents, b.tangents, (n) => new Float32Array(n)),
    joints: join(a.joints, b.joints, (n) => new Uint8Array(n)),
    weights: join(a.weights, b.weights, (n) => new Float32Array(n)),
    sections: Uint8Array.from([...(a.sections ?? new Uint8Array(a.indices.length / 3)), ...[...(b.sections ?? new Uint8Array(b.indices.length / 3))].map((s) => s + body.materials!.length)]),
  };
  const shift = body.textures!.length;
  const theirs = saddle.materials.map((m) => {
    const moved = { ...m };
    for (const role of TEXTURE_ROLES) if (textureOf(m, role) >= 0) setTexture(moved, role, textureOf(m, role) + shift);
    return moved;
  });
  const mesh: SourceMesh = { ...body, id: "core:pteranodon", lods: [lod], flat: 1, materials: [...body.materials, ...theirs], textures: [...body.textures!, ...saddle.textures!] };
  return parsePack(buildPack([mesh], pack.palette, pack.clips.clips, pack.header.rigs));
}
const saddled = saddledPack();
const saddledTextures = saddled ? texturesOf(saddled) : [];

const hex = (h: string): number => nearestColour(pack.palette, parseInt(h.slice(0, 2), 16), parseInt(h.slice(2, 4), 16), parseInt(h.slice(4, 6), 16));
const DYES: Record<string, readonly number[]> = {

const GROUND = [0.84, 0.83, 0.81] as const;
const SKY = [0.74, 0.82, 0.92] as const;
// Point looked at (animal frame: x left, y up, z forward), radius held, degrees around from the front and above.
type Shot = [centre: readonly [number, number, number], radius: number, around: number, above: number, sky: boolean];
const SHOTS: Record<string, Shot> = {
  below: [[0, 0.8, -0.2], 3.6, 0, -82, true],
  above: [[0, 0.8, -0.2], 3.6, 180, 82, false],
  three_quarter: [[0, 0.85, 0], 2.2, 35, -12, true],
  side: [[0, 0.95, 0.1], 1.6, 90, 4, false],
  head: [[0, 1.45, 0.85], 0.75, 80, 6, false],
  head_front: [[0, 1.4, 0.9], 0.55, 15, 8, false],
  rider: [[0, 1.15, 0.35], 0.75, 185, 32, false],
  inner_wing: [[1.1, 0.85, -0.35], 1.1, 160, 45, false],
  under_wing: [[1.2, 0.8, -0.35], 1.1, 10, -65, true],
  feet: [[0, 0.45, -1.05], 0.35, 60, -20, false],
  // With its saddle: the rider's view, the seat close, and the whole from the side.
  saddled_rider: [[0, 1.1, 0.25], 0.75, 185, 32, false],
  saddled_seat: [[0, 1.0, -0.1], 0.4, 140, 35, false],
  saddled_side: [[0, 1.0, 0.2], 1.0, 80, 10, false],
  saddled_front: [[0, 1.0, 0.3], 0.6, 25, 15, false],

const W = 1200;
const H = 900;
const SS = 2;

function draw(view: ModelView, shot: Shot, colors: readonly number[], mesh: number): void {
  const [centre, radius, around, above] = shot;
  view.draw({ mesh, level: 0, around, above, centre, radius, rig, colors, textured: true });
}

function viewOf(w: number, h: number, sky: boolean, withSaddle = false): ModelView {
  const v = new ModelView(withSaddle ? saddled! : pack, w, h, sky ? SKY : GROUND);
  (withSaddle ? saddledTextures : textures).forEach((t, slot) => v.texture(slot, t.role, t.pictures));
  return v;
}

// From below against the sky at the distances it is seen: its span (6.96 m) about 270, 90, 36 and 13 pixels across on
// the game's 1,080-row screen, each drawn at that size so the renderer reads the mips and levels it would, then set
// side by side at that size and enlarged four times. Levels are chosen by the size of the picture, as the game does.
function sizes(colors: readonly number[]): { w: number; h: number; img: Uint8Array } {
  const pad = 12;
  const spans = [270, 90, 36, 13];
  const pieces = spans.map((span) => {
    const sw = Math.round((span * 7.2) / 6.96);
    const sh = Math.max(4, Math.round(sw / 2));
    const v = viewOf(sw, sh, true);
    const [centre, , around, above] = SHOTS.below!;
    v.draw({ mesh: index, level: 0, around, above, centre, radius: 3.6, rig, colors, textured: true });
    return { small: v.raster.color.slice(), sw, sh };
  });
  const w = pieces.reduce((s, p) => s + p.sw * 5 + pad * 2, pad);
  const h = pieces[0]!.sh * 4 + pad * 2;
  const img = new Uint8Array(w * h * 4).fill(214);
  let x = pad;
  for (const { small, sw, sh } of pieces) {
    for (let y = 0; y < sh; y++) for (let k = 0; k < sw * 4; k++) img[((pad + y) * w) * 4 + x * 4 + k] = small[y * sw * 4 + k]!;
    x += sw + pad;
  }
  for (const { small, sw, sh } of pieces) {
    for (let y = 0; y < sh * 4; y++) for (let xx = 0; xx < sw * 4; xx++) for (let c = 0; c < 4; c++) img[((pad + y) * w + x + xx) * 4 + c] = small[(Math.floor(y / 4) * sw + Math.floor(xx / 4)) * 4 + c]!;
    x += sw * 4 + pad;
  }
  return { w, h, img };
}

const STAGES = Array.from({ length: LAYER_STAGES + 1 }, (_, s) => (s * LAYER_LEVELS) / LAYER_STAGES);
if (args.includes("--stages")) {
  const side = 300;
  for (const [dyes, colors] of Object.entries(DYES)) {
    for (const name of ["three_quarter", "above", "side", "head", "rider", "inner_wing", "under_wing"]) {
      const shot = SHOTS[name]!;
      const pad = 8;
      const w = STAGES.length * (side + pad) + pad;
      const h = side + pad * 2;
      const img = new Uint8Array(w * h * 4).fill(214);
      const view = viewOf(side * SS, side * SS, shot[4]);
      STAGES.forEach((layer, k) => {
        view.draw({ mesh: index, level: 0, around: shot[2], above: shot[3], centre: shot[0], radius: shot[1], rig, colors, textured: true, layer });
        const src = view.raster.color;
        const x0 = pad + k * (side + pad);
        for (let y = 0; y < side; y++) {
          for (let x = 0; x < side; x++) {
            for (let c = 0; c < 4; c++) {
              let sum = 0;
              for (let dy = 0; dy < SS; dy++) for (let dx = 0; dx < SS; dx++) sum += src[((y * SS + dy) * side * SS + x * SS + dx) * 4 + c]!;
              img[((pad + y) * w + x0 + x) * 4 + c] = Math.round(sum / (SS * SS));
            }
          }
        }
      });
      const path = join(out, `stages_${name}_${dyes}.png`);
      writeFileSync(path, encodePng(w, h, img));
      console.log(path);
    }
  }
  process.exit(0);
}

for (const [dyes, colors] of Object.entries(DYES)) {
  if (dyes !== "design") continue;
  const s = sizes(colors);
  const file = join(out, `sizes_${dyes}.png`);
  writeFileSync(file, encodePng(s.w, s.h, s.img));
  console.log(file);
  for (const [name, shot] of Object.entries(SHOTS)) {
    if (dyes !== "design" && !["below", "three_quarter", "head", "rider", "saddled_rider"].includes(name)) continue;
    if (!["saddled_front", "saddled_seat", "saddled_rider"].includes(name)) continue;
    const withSaddle = name.startsWith("saddled");
    if (withSaddle && !saddled) continue;
    const view = viewOf(W * SS, H * SS, shot[4], withSaddle);
    draw(view, shot, colors, withSaddle ? 0 : index);
    const src = view.raster.color;
    const img = new Uint8Array(W * H * 4);
    for (let y = 0; y < H; y++) {
      for (let x = 0; x < W; x++) {
        for (let c = 0; c < 4; c++) {
          let sum = 0;
          for (let dy = 0; dy < SS; dy++) for (let dx = 0; dx < SS; dx++) sum += src[((y * SS + dy) * W * SS + x * SS + dx) * 4 + c]!;
          img[(y * W + x) * 4 + c] = Math.round(sum / (SS * SS));
        }
      }
    }
    const path = join(out, `${name}_${dyes}.png`);
    writeFileSync(path, encodePng(W, H, img));
    console.log(path);
  }
}
